# Memiarka Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Zsynchronizowane „radio memowe" — serwer Node/TS steruje odtwarzaniem zdjęć/filmów u wszystkich klientów jednocześnie, z reklamami, trybami specjalnymi i panelem kontrolnym.

**Architecture:** Jeden proces Node.js: Express (statyki + media z Range) + WebSocket (`ws`). Serwer trzyma jedyny autorytatywny stan odtwarzania (`Radio`); klienci są „głupimi" renderami sterowanymi komunikatami `state` i synchronizują zegar pingami. Frontend: dwie strony vanilla TS budowane Vitem — panel na `/`, odtwarzacz na `/player`.

**Tech Stack:** Node 22+, TypeScript (strict), Express, ws, chokidar, Vite (MPA), Vitest, tsx (runtime TS, także w prod), Docker (node:22-alpine + ffmpeg z apk).

**Spec:** `docs/superpowers/specs/2026-08-06-memiarka-design.md`

## Global Constraints

- Node >= 22, npm, TypeScript strict.
- `ffprobe` musi być w `PATH` (dev: systemowy ffmpeg; Docker: `apk add ffmpeg`). Żadnej zmiennej `FFPROBE_PATH`.
- URL-e trybu `pantry` zahardkodowane: `http://czyjestser.www22.duck.edu.pl/ser.txt`, `http://czyjestmleko.www22.duck.edu.pl/mleko.txt`, `http://czyjestchlebtostowy.www22.duck.edu.pl/chleb.txt`.
- Env + domyślne: `PORT=8080`, `MEDIA_DIR=./media`, `ADS_DIR=./ads`, `IMAGE_DURATION=10` (s), `AD_INTERVAL=10`, `SPECIAL_INTERVAL=300` (s), `HISTORY_SIZE=50`.
- Teksty UI po polsku, dokładnie: `NOWY MEM!`, `wrzuć pliki do ./media`, `▶ START`, `Pełny ekran`, `Odtwórz reklamę teraz`, `Przerwij odtwarzanie reklamy`.
- Rozszerzenia: obrazki `.jpg .jpeg .png .gif .webp`, wideo `.mp4 .webm`. Katalog reklam: tylko wideo. Bez podkatalogów (depth 0).
- Priorytet przy przejściu do następnego elementu: kolejka NOWY MEM → gotowy tryb specjalny → reklama (licznik) → mem. NOWY MEM przerywa natychmiast wszystko POZA reklamą (wtedy kolejka FIFO).
- Testy: Vitest, `npm test` musi przechodzić po każdym tasku. Commit po każdym tasku.

```ts
export function mulberry32(seed: number): () => number {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
```

`src/server/shuffle-bag.ts`:
```ts
export class ShuffleBag<T> {
  private pool: T[] = [];
  private bag: T[] = [];
  private last: T | undefined;

  constructor(private rng: () => number = Math.random) {}

  get size(): number {
    return this.pool.length;
  }

  setItems(items: T[]): void {
    const next = new Set(items);
    const old = new Set(this.pool);
    this.bag = this.bag.filter((x) => next.has(x));
    for (const item of items) {
      if (!old.has(item)) {
        const i = Math.floor(this.rng() * (this.bag.length + 1));
        this.bag.splice(i, 0, item);
      }
    }
    this.pool = [...items];
  }

  next(): T | undefined {
    if (this.pool.length === 0) return undefined;
    if (this.bag.length === 0) {
      this.bag = [...this.pool];
      for (let i = this.bag.length - 1; i > 0; i--) {
        const j = Math.floor(this.rng() * (i + 1));
        [this.bag[i], this.bag[j]] = [this.bag[j], this.bag[i]];
      }
      // next() zdejmuje z końca — nie powtarzaj ostatnio zagranego na styku cykli
      if (this.pool.length > 1 && this.bag[this.bag.length - 1] === this.last) {
        [this.bag[0], this.bag[this.bag.length - 1]] = [this.bag[this.bag.length - 1], this.bag[0]];
      }
    }
    this.last = this.bag.pop();
    return this.last;
  }
}
```

```ts
import { execFile } from 'node:child_process';

export function parseDuration(output: string): number {
  const seconds = Number.parseFloat(output.trim());
  if (!Number.isFinite(seconds) || seconds <= 0) {
    throw new Error(`ffprobe: nieprawidłowa długość: ${JSON.stringify(output)}`);
  }
  return Math.round(seconds * 1000);
}

export function probeDuration(file: string): Promise<number> {
  return new Promise((resolve, reject) => {
    execFile(
      'ffprobe',
      ['-v', 'error', '-show_entries', 'format=duration', '-of', 'csv=p=0', file],
      (err, stdout) => {
        if (err) return reject(err);
        try {
          resolve(parseDuration(stdout));
        } catch (e) {
          reject(e);
        }
      },
    );
  });
}
```

```ts
  items(): MediaItem[] {
    return [...this.byFile.values()];
  }

  get(file: string): MediaItem | undefined {
    return this.byFile.get(file);
  }

  async start(): Promise<void> {
    // Skan początkowy własnymi siłami — chokidarowe eventy `add` z awaitWriteFinish
    // potrafią przyjść po `ready` i skan byłby niedeterministyczny.
    const entries = await readdir(this.dir, { withFileTypes: true });
    await Promise.all(
      entries.filter((e) => e.isFile()).map((e) => this.tryAdd(join(this.dir, e.name), false)),
    );
    this.watcher = chokidar.watch(this.dir, {
      depth: 0,
      ignoreInitial: true,
      awaitWriteFinish: {
        stabilityThreshold: this.opts.stabilityThreshold ?? 2000,
        pollInterval: 100,
      },
    });
    this.watcher.on('add', (p) => void this.tryAdd(p, true));
    this.watcher.on('unlink', (p) => {
      const file = basename(p);
      if (this.byFile.delete(file)) this.emit('removed', file);
    });
    await new Promise<void>((res) => this.watcher!.once('ready', res));
  }

  async stop(): Promise<void> {
    await this.watcher?.close();
  }

  private async tryAdd(p: string, emit: boolean): Promise<void> {
    const ext = extname(p).toLowerCase();
    let type: 'image' | 'video';
    if (VIDEO_EXT.has(ext)) type = 'video';
    else if (IMAGE_EXT.has(ext) && !this.opts.videosOnly) type = 'image';
    else return;
    try {
      const duration = type === 'video' ? await this.probe(p) : undefined;
      const item: MediaItem = { file: basename(p), path: p, type, duration };
      this.byFile.set(item.file, item);
      if (emit) this.emit('added', item);
    } catch (err) {
      console.warn(`[library] pomijam ${p}:`, err instanceof Error ? err.message : err);
    }
  }
}
```

```ts
import { EventEmitter } from 'node:events';
import type { HistoryEntry, StateItem, StateMessage } from '../shared/types';
import type { MediaItem, MediaSource } from './library';
import { ShuffleBag } from './shuffle-bag';
import type { SpecialMode } from './specials/index';

export interface RadioConfig {
  imageDurationMs: number;
  adInterval: number;
  specialIntervalMs: number;
  historySize: number;
}

interface RadioDeps {
  memes: MediaSource;
  ads: MediaSource;
  specials: SpecialMode[];
  config: RadioConfig;
  rng?: () => number;
}

interface CurrentState {
  item: StateItem;
  startedAt: number;
  duration: number;
  newMem: boolean;
}

export class Radio extends EventEmitter {
  private memes: MediaSource;
  private ads: MediaSource;
  private specials: SpecialMode[];
  private config: RadioConfig;
  private rng: () => number;

  private memeBag: ShuffleBag<string>;
  private adBag: ShuffleBag<string>;
  private current: CurrentState = { item: { kind: 'idle' }, startedAt: 0, duration: 0, newMem: false };
  private historyBuf: HistoryEntry[] = [];
  private memesSinceAd = 0;
  private newMemQueue: MediaItem[] = [];
  private pendingSpecial: { mode: SpecialMode; data: unknown } | null = null;
  private fetchingSpecial = false;
  private specialIndex = 0;
  private endTimer: ReturnType<typeof setTimeout> | undefined;
  private specialTimer: ReturnType<typeof setInterval> | undefined;

  constructor(deps: RadioDeps) {
    super();
    this.memes = deps.memes;
    this.ads = deps.ads;
    this.specials = deps.specials;
    this.config = deps.config;
    this.rng = deps.rng ?? Math.random;
    this.memeBag = new ShuffleBag<string>(this.rng);
    this.adBag = new ShuffleBag<string>(this.rng);
  }

  start(): void {
    this.syncBags();
    this.memes.on('added', (item: MediaItem) => this.onNewMeme(item));
    this.memes.on('removed', () => this.syncBags());
    this.ads.on('added', () => { this.syncBags(); this.emit('update'); });
    this.ads.on('removed', () => { this.syncBags(); this.emit('update'); });
    if (this.specials.length > 0) {
      this.specialTimer = setInterval(() => this.prefetchSpecial(), this.config.specialIntervalMs);
    }
    this.advance();
  }

  stop(): void {
    clearTimeout(this.endTimer);
    clearInterval(this.specialTimer);
  }

  stateMessage(): StateMessage {
    return {
      type: 'state',
      ...this.current,
      serverNow: Date.now(),
      adsAvailable: this.adBag.size > 0,
    };
  }

  history(): HistoryEntry[] {
    return [...this.historyBuf];
  }

  playAdNow(): void {
    const item = this.nextAd();
    if (item) this.playMedia(item, { ad: true });
  }

  stopAd(): void {
    if (this.current.item.kind !== 'ad') return;
    this.advance();
  }

  private syncBags(): void {
    this.memeBag.setItems(this.memes.items().map((i) => i.file));
    this.adBag.setItems(this.ads.items().map((i) => i.file));
  }

  private onNewMeme(item: MediaItem): void {
    this.syncBags();
    if (this.current.item.kind === 'ad') {
      this.newMemQueue.push(item);
      return;
    }
    this.playMedia(item, { newMem: true });
  }

  private prefetchSpecial(): void {
    if (this.specials.length === 0 || this.pendingSpecial || this.fetchingSpecial) return;
    const mode = this.specials[this.specialIndex % this.specials.length];
    this.specialIndex++;
    this.fetchingSpecial = true;
    void mode
      .fetchData()
      .catch(() => null)
      .then((data) => {
        this.fetchingSpecial = false;
        this.pendingSpecial = { mode, data };
      });
  }

  private advance(): void {
    const queued = this.newMemQueue.shift();
    if (queued) return this.playMedia(queued, { newMem: true });

    if (this.pendingSpecial) {
      const { mode, data } = this.pendingSpecial;
      this.pendingSpecial = null;
      return this.playSpecial(mode, data);
    }

    if (this.memesSinceAd >= this.config.adInterval) {
      const ad = this.nextAd();
      if (ad) return this.playMedia(ad, { ad: true });
    }

    const file = this.memeBag.next();
    const item = file ? this.memes.get(file) : undefined;
    if (!item) return this.playIdle();
    this.memesSinceAd++;
    this.playMedia(item, {});
  }

  private nextAd(): MediaItem | undefined {
    const file = this.adBag.next();
    const item = file ? this.ads.get(file) : undefined;
    if (item) this.memesSinceAd = 0;
    return item;
  }

  private playMedia(item: MediaItem, opts: { newMem?: boolean; ad?: boolean }): void {
    const kind = opts.ad ? 'ad' : item.type;
    const duration = item.type === 'image' ? this.config.imageDurationMs : (item.duration ?? this.config.imageDurationMs);
    const url = `${opts.ad ? '/ads' : '/media'}/${encodeURIComponent(item.file)}`;
    this.setState({ kind, url, name: item.file }, duration, opts.newMem ?? false);
  }

  private playSpecial(mode: SpecialMode, data: unknown): void {
    this.setState(
      { kind: 'special', specialId: mode.id, name: mode.id, data, seed: Math.floor(this.rng() * 2 ** 31) },
      mode.duration,
      false,
    );
  }

  private playIdle(): void {
    clearTimeout(this.endTimer);
    this.current = { item: { kind: 'idle' }, startedAt: Date.now(), duration: 0, newMem: false };
    this.emit('update');
  }

  private setState(item: StateItem, duration: number, newMem: boolean): void {
    clearTimeout(this.endTimer);
    this.current = { item, startedAt: Date.now(), duration, newMem };
    this.historyBuf.unshift({
      kind: item.kind,
      name: item.name ?? item.specialId ?? '?',
      url: item.url,
      playedAt: Date.now(),
      newMem,
    });
    if (this.historyBuf.length > this.config.historySize) this.historyBuf.pop();
    this.endTimer = setTimeout(() => this.advance(), duration);
    this.emit('update');
  }
}
```

```ts
import type { ClientMessage, HistoryMessage } from '../shared/types';
import type { Config } from './config';
import { MediaLibrary } from './library';
import { Radio } from './radio';
import { specials } from './specials/index';

export interface App {
  server: http.Server;
  radio: Radio;
  stop(): Promise<void>;
}

export async function createApp(config: Config): Promise<App> {
  mkdirSync(config.mediaDir, { recursive: true });
  mkdirSync(config.adsDir, { recursive: true });

  const memes = new MediaLibrary(config.mediaDir);
  const ads = new MediaLibrary(config.adsDir, { videosOnly: true });
  await Promise.all([memes.start(), ads.start()]);

  const radio = new Radio({
    memes,
    ads,
    specials,
    config: {
      imageDurationMs: config.imageDurationMs,
      adInterval: config.adInterval,
      specialIntervalMs: config.specialIntervalMs,
      historySize: config.historySize,
    },
  });

  const distClient = join(process.cwd(), 'dist/client');
  const app = express();
  app.use(express.static(distClient));
  app.get('/player', (_req, res) => res.sendFile(join(distClient, 'player.html')));
  app.use('/media', express.static(config.mediaDir));
  app.use('/ads', express.static(config.adsDir));

  const server = http.createServer(app);
  const wss = new WebSocketServer({ server, path: '/ws' });

  const historyMessage = (): HistoryMessage => ({ type: 'history', entries: radio.history() });
  const broadcast = (msg: unknown): void => {
    const raw = JSON.stringify(msg);
    for (const client of wss.clients) {
      if (client.readyState === WebSocket.OPEN) client.send(raw);
    }
  };

  wss.on('connection', (ws) => {
    ws.send(JSON.stringify(radio.stateMessage()));
    ws.send(JSON.stringify(historyMessage()));
    ws.on('message', (raw) => {
      let msg: ClientMessage;
      try {
        msg = JSON.parse(String(raw));
      } catch {
        return;
      }
      if (msg.type === 'ping') {
        ws.send(JSON.stringify({ type: 'pong', t0: msg.t0, serverTime: Date.now() }));
      } else if (msg.type === 'playAd') {
        radio.playAdNow();
      } else if (msg.type === 'stopAd') {
        radio.stopAd();
      }
    });
  });

  radio.on('update', () => {
    broadcast(radio.stateMessage());
    broadcast(historyMessage());
  });
  radio.start();

  return {
    server,
    radio,
    async stop(): Promise<void> {
      radio.stop();
      wss.close();
      await new Promise<void>((res) => server.close(() => res()));
      await Promise.all([memes.stop(), ads.stop()]);
    },
  };
}
```

```ts
const MAX_SAMPLES = 7;

export function computeOffset(t0: number, serverTime: number, t1: number): number {
  return serverTime - (t0 + t1) / 2;
}

export function median(xs: number[]): number {
  if (xs.length === 0) throw new Error('median: pusta tablica');
  const s = [...xs].sort((a, b) => a - b);
  const mid = Math.floor(s.length / 2);
  return s.length % 2 === 1 ? s[mid] : (s[mid - 1] + s[mid]) / 2;
}

export class SyncedClock {
  private samples: number[] = [];

  addSample(t0: number, serverTime: number, t1: number): void {
    this.samples.push(computeOffset(t0, serverTime, t1));
    if (this.samples.length > MAX_SAMPLES) this.samples.shift();
  }

  get offset(): number {
    return this.samples.length === 0 ? 0 : median(this.samples);
  }

  now(): number {
    return Date.now() + this.offset;
  }
}
```

```ts
import type { ClientMessage, HistoryMessage, ServerMessage, StateMessage } from '../../shared/types';
import { SyncedClock } from './clock';

export interface Connection {
  send(msg: ClientMessage): void;
  clock: SyncedClock;
}

const PING_INTERVAL_MS = 10_000;
const BACKOFF_MS = [1000, 2000, 4000, 8000, 10_000];

export function connect(onMessage: (msg: StateMessage | HistoryMessage) => void): Connection {
  const clock = new SyncedClock();
  let ws: WebSocket | null = null;
  let attempt = 0;
  let pingTimer: ReturnType<typeof setInterval> | undefined;

  function open(): void {
    const proto = location.protocol === 'https:' ? 'wss' : 'ws';
    ws = new WebSocket(`${proto}://${location.host}/ws`);
    ws.onopen = () => {
      attempt = 0;
      sendPing();
      pingTimer = setInterval(sendPing, PING_INTERVAL_MS);
    };
    ws.onmessage = (ev) => {
      const msg = JSON.parse(ev.data) as ServerMessage;
      if (msg.type === 'pong') clock.addSample(msg.t0, msg.serverTime, Date.now());
      else onMessage(msg);
    };
    ws.onclose = () => {
      clearInterval(pingTimer);
      const delay = BACKOFF_MS[Math.min(attempt, BACKOFF_MS.length - 1)];
      attempt++;
      setTimeout(open, delay);
    };
    ws.onerror = () => ws?.close();
  }

  function sendPing(): void {
    send({ type: 'ping', t0: Date.now() });
  }

  function send(msg: ClientMessage): void {
    if (ws?.readyState === WebSocket.OPEN) ws.send(JSON.stringify(msg));
  }

  open();
  return { send, clock };
}
```

```ts
const canvas = $<HTMLCanvasElement>('#layer-canvas');

const banner = $('#banner');
const videoError = $('#video-error');
const startOverlay = $('#start-overlay');

const BANNER_MS = 4000;
let started = false;
let lastState: StateMessage | null = null;
let rafId = 0;
let bannerTimer: ReturnType<typeof setTimeout> | undefined;

const conn = connect((msg) => {
  if (msg.type === 'state') {
    lastState = msg;
    if (started) apply(msg);
  }
});

$('#start').addEventListener('click', () => {
  started = true;
  startOverlay.classList.add('hidden');
  if (lastState) apply(lastState);
});
$('#fullscreen').addEventListener('click', () => {
  void document.documentElement.requestFullscreen().catch(() => {});
});

function elapsed(state: StateMessage): number {
  return conn.clock.now() - state.startedAt;
}

function hideAll(): void {
  cancelAnimationFrame(rafId);
  for (const el of [img, video, canvas]) el.classList.remove('visible');
  idleEl.classList.add('hidden');
  videoError.classList.add('hidden');
  video.pause();
  video.removeAttribute('src');
  video.load();
}

function apply(state: StateMessage): void {
  hideAll();
  switch (state.item.kind) {
    case 'idle':
      idleEl.classList.remove('hidden');
      break;
    case 'image':
      img.src = state.item.url!;
      img.classList.add('visible');
      break;
    case 'video':
    case 'ad':
      showVideo(state);
      break;
    case 'special':
      showSpecial(state);
      break;
  }
  clearTimeout(bannerTimer);
  if (state.newMem && elapsed(state) < BANNER_MS) {
    banner.classList.remove('hidden');
    bannerTimer = setTimeout(() => banner.classList.add('hidden'), BANNER_MS - elapsed(state));
  } else {
    banner.classList.add('hidden');
  }
}

function showVideo(state: StateMessage): void {
  const fail = (): void => {
    hideAll();
    videoError.textContent = state.item.name ?? '';
    videoError.classList.remove('hidden');
  };
  video.src = state.item.url!;
  video.addEventListener(
    'loadedmetadata',
    () => {
      video.currentTime = Math.max(0, elapsed(state) / 1000);
      video.play().catch(fail);
    },
    { once: true },
  );
  video.addEventListener('error', fail, { once: true });
  video.classList.add('visible');
}

function showSpecial(state: StateMessage): void {
  const special = clientSpecials.find((s) => s.id === state.item.specialId);
  if (!special) return; // nieznany tryb — czarny ekran do końca slotu
  canvas.classList.add('visible');
  const ctx = canvas.getContext('2d')!;
  const seed = state.item.seed ?? 0;
  const frame = (): void => {
    canvas.width = canvas.clientWidth;
    canvas.height = canvas.clientHeight;
    special.render(ctx, elapsed(state), canvas.width, canvas.height, state.item.data, mulberry32(seed));
    rafId = requestAnimationFrame(frame);
  };
  rafId = requestAnimationFrame(frame);
}

// Tryb podglądu trybów specjalnych: /player?preview=pantry (bez WS, dane z previewData)
const preview = new URLSearchParams(location.search).get('preview');
if (preview) {
  const special = clientSpecials.find((s) => s.id === preview);
  startOverlay.classList.add('hidden');
  started = true;
  if (special) {
    const fakeState: StateMessage = {
      type: 'state',
      item: { kind: 'special', specialId: special.id, name: special.id, data: special.previewData, seed: 42 },
      startedAt: Date.now(),
      duration: 999_999,
      newMem: false,
      serverNow: Date.now(),
      adsAvailable: false,
    };
    apply(fakeState);
  }
}
```

```ts
import type { PantryData, PantryItemState } from '../../shared/types';
import type { ClientSpecial } from './index';

const ROWS: { key: keyof PantryData; emoji: string; label: string }[] = [
  { key: 'ser', emoji: '🧀', label: 'SER' },
  { key: 'mleko', emoji: '🥛', label: 'MLEKO' },
  { key: 'chleb', emoji: '🍞', label: 'CHLEB TOSTOWY' },
];

const COLORS = { ok: '#3fbf5f', low: '#e6c229', out: '#e63946', unknown: '#888888' } as const;
const UNKNOWN: PantryItemState = { status: 'unknown', text: '???' };

export const pantrySpecial: ClientSpecial = {
  id: 'pantry',
  previewData: {
    ser: { status: 'ok', text: 'Otóż TAK!!!' },
    mleko: { status: 'low', text: 'Już prawie nie ma' },
    chleb: { status: 'out', text: 'GŁODÓWKA NIE MA TOSTÓW!!!!!!!!!!!!1111111jedenjeden' },
  } satisfies PantryData,

  render(ctx, t, w, h, rawData): void {
    const data = rawData as PantryData | null;
    ctx.fillStyle = '#101018';
    ctx.fillRect(0, 0, w, h);

    const anyOut = data !== null && ROWS.some((r) => data[r.key].status === 'out');
    ctx.save();
    if (anyOut) ctx.translate(Math.sin(t / 30) * 6, Math.cos(t / 37) * 6);

    ctx.textAlign = 'center';
    ctx.textBaseline = 'middle';
    ctx.fillStyle = '#ffffff';
    ctx.font = `bold ${h * 0.08}px system-ui, sans-serif`;
    ctx.fillText('CZY JEST...?', w / 2, h * 0.11);

    ROWS.forEach((row, i) => {
      const y = h * (0.3 + i * 0.24);
      const st = data ? data[row.key] : UNKNOWN;
      const color = COLORS[st.status];

      if (st.status === 'out' && Math.sin(t / 80) > 0) {
        ctx.fillStyle = 'rgba(230, 57, 70, 0.25)';
        ctx.fillRect(0, y - h * 0.1, w, h * 0.2);
      }
      ctx.globalAlpha = st.status === 'low' ? (Math.sin(t / 150) > 0 ? 1 : 0.35) : 1;

      const pulse = st.status === 'ok' ? 1 + 0.05 * Math.sin(t / 300 + i * 2) : 1;
      ctx.font = `${h * 0.11 * pulse}px system-ui, sans-serif`;
      ctx.fillText(st.status === 'unknown' ? '❓' : row.emoji, w * 0.14, y);

      ctx.textAlign = 'left';
      ctx.fillStyle = color;
      ctx.font = `bold ${h * 0.05}px system-ui, sans-serif`;
      ctx.fillText(row.label, w * 0.24, y - h * 0.035);
      ctx.fillStyle = '#ffffff';
      ctx.font = `${h * 0.032}px system-ui, sans-serif`;
      ctx.fillText(st.text, w * 0.24, y + h * 0.045, w * 0.72);
      ctx.textAlign = 'center';
      ctx.globalAlpha = 1;
    });
    ctx.restore();
  },
};
```

```ts
import type { HistoryEntry, ItemKind, StateMessage } from '../../shared/types';
import { connect } from '../lib/connection';

const $ = <T extends HTMLElement>(sel: string): T => document.querySelector(sel)!;
const nowContent = $('#now-content');
const historyList = $<HTMLUListElement>('#history');
const playAdBtn = $<HTMLButtonElement>('#play-ad');
const stopAdBtn = $<HTMLButtonElement>('#stop-ad');
const limitInput = $<HTMLInputElement>('#limit');

const KIND_LABEL: Record<ItemKind, string> = {
  image: 'mem', video: 'wideo', ad: 'reklama', special: 'tryb specjalny', idle: 'bezczynność',
};
const KIND_ICON: Record<ItemKind, string> = {
  image: '🖼', video: '▶', ad: '📢', special: '✨', idle: '💤',
};

let state: StateMessage | null = null;
let history: HistoryEntry[] = [];

limitInput.value = localStorage.getItem('memiarka-limit') ?? '20';
limitInput.addEventListener('change', () => {
  localStorage.setItem('memiarka-limit', limitInput.value);
  renderHistory();
});

const conn = connect((msg) => {
  if (msg.type === 'state') {
    state = msg;
    renderNow();
  } else {
    history = msg.entries;
    renderHistory();
  }
});

playAdBtn.addEventListener('click', () => conn.send({ type: 'playAd' }));
stopAdBtn.addEventListener('click', () => conn.send({ type: 'stopAd' }));

function thumb(kind: ItemKind, url?: string): HTMLElement {
  if (kind === 'image' && url) {
    const el = document.createElement('img');
    el.src = url;
    return el;
  }
  const el = document.createElement('span');
  el.className = 'thumb-icon';
  el.textContent = KIND_ICON[kind];
  return el;
}

function renderNow(): void {
  if (!state) return;
  nowContent.replaceChildren();
  const { item } = state;
  if (item.kind === 'idle') {
    nowContent.textContent = 'nic nie gra — wrzuć pliki do ./media';
  } else {
    nowContent.append(thumb(item.kind, item.url));
    const meta = document.createElement('div');
    meta.className = 'meta';
    const name = document.createElement('span');
    name.className = 'name';
    name.textContent = item.name ?? '?';
    const kind = document.createElement('span');
    kind.className = 'kind';
    kind.textContent = KIND_LABEL[item.kind];
    meta.append(name, kind);
    if (state.newMem) {
      const badge = document.createElement('span');
      badge.className = 'badge';
      badge.textContent = 'NOWY MEM!';
      meta.append(badge);
    }
    nowContent.append(meta);
  }
  playAdBtn.disabled = !state.adsAvailable;
  stopAdBtn.disabled = state.item.kind !== 'ad';
}

function renderHistory(): void {
  const limit = Math.max(1, Number(limitInput.value) || 20);
  historyList.replaceChildren(
    ...history.slice(0, limit).map((e) => {
      const li = document.createElement('li');
      li.append(thumb(e.kind, e.url));
      const meta = document.createElement('div');
      meta.className = 'meta';
      const name = document.createElement('span');
      name.className = 'name';
      name.textContent = e.name;
      const kind = document.createElement('span');
      kind.className = 'kind';
      kind.textContent = KIND_LABEL[e.kind];
      meta.append(name, kind);
      if (e.newMem) {
        const badge = document.createElement('span');
        badge.className = 'badge';
        badge.textContent = 'NOWY MEM!';
        meta.append(badge);
      }
      const time = document.createElement('span');
      time.className = 'time';
      time.textContent = new Date(e.playedAt).toLocaleTimeString('pl-PL');
      li.append(meta, time);
      return li;
    }),
  );
}
```
